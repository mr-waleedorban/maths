// ===== AUTH =====
function signInWithGoogle() {
    auth.signInWithPopup(googleProvider)
        .catch(err => showError(err.message));
}

function signInWithEmail() {
    const email = document.getElementById('email-input').value;
    const password = document.getElementById('password-input').value;
    if (!email || !password) return showError('اكمل البيانات');
    auth.signInWithEmailAndPassword(email, password)
        .catch(err => showError('البريد أو كلمة المرور غير صحيحة'));
}

function signUpWithEmail() {
    const email = document.getElementById('email-input').value;
    const password = document.getElementById('password-input').value;
    if (!email || !password) return showError('اكمل البيانات');
    if (password.length < 6) return showError('كلمة المرور لازم 6 حروف على الأقل');
    
    auth.createUserWithEmailAndPassword(email, password)
        .catch(err => showError(err.message));
}

function logout() {
    showScreen('loading-screen');
    auth.signOut().then(() => {
        currentUser = null;
        showScreen('login-screen');
    });
}

function showError(msg) {
    const el = document.getElementById('login-error');
    el.textContent = msg;
    el.classList.remove('hidden');
    setTimeout(() => el.classList.add('hidden'), 4000);
}

        const userEmail = user.email.toLowerCase();
        
        // فحص هل الإيميل هو إيميل الأدمن المخصص
        const assignedRole = (userEmail === ADMIN_EMAIL.toLowerCase()) ? 'teacher' : 'student';

        db.collection('users').doc(user.uid).get().then(doc => {
            if (doc.exists) {
                const data = doc.data();
                
                // تحديث الـ Role لو تحول البريد لأدمن
                if (data.role !== assignedRole) {
                    db.collection('users').doc(user.uid).update({ role: assignedRole });
                    data.role = assignedRole;
                }
                
                currentRole = data.role;
                handleAuthSuccess(user, data);
            } else {
                // إنشاء حساب جديد مع تعيين الـ Role الصحيح
                const newUser = {
                    name: user.displayName || user.email.split('@')[0],
                    email: user.email,
                    role: assignedRole,
                    points: 0,
                    progress: 0,
                    lessonsCompleted: 0,
                    examsTaken: 0,
                    photoURL: user.photoURL || '',
                    createdAt: firebase.firestore.FieldValue.serverTimestamp()
                };
                
                db.collection('users').doc(user.uid).set(newUser).then(() => {
                    currentRole = assignedRole;
                    handleAuthSuccess(user, newUser);
                });
            }
        }).catch(err => {
            console.error("Error fetching user role:", err);
            showError("حدث خطأ في جلب بيانات الحساب");
        });
    } else {
        showScreen('login-screen');
    }
});

function handleAuthSuccess(user, data) {
    document.getElementById('login-error').classList.add('hidden');
    if (currentRole === 'teacher') {
        showScreen('teacher-dashboard');
        loadTeacherData();
    } else {
        showScreen('student-dashboard');
        loadStudentData(user.uid);
    }
}

function showScreen(id) {
    ['loading-screen', 'login-screen', 'student-dashboard', 'teacher-dashboard'].forEach(s => {
        document.getElementById(s).classList.add('hidden');
    });
    document.getElementById(id).classList.remove('hidden');
}

// ===== TEACHER FUNCTIONS =====
function loadTeacherData() {
    db.collection('users').where('role', '==', 'student').onSnapshot(snap => {
        allStudents = [];
        let total = 0, active = 0, totalPoints = 0;
        const tbody = document.getElementById('teacher-students-table');
        tbody.innerHTML = '';
        
        snap.forEach(doc => {
            const s = doc.data();
            s.id = doc.id;
            allStudents.push(s);
            total++;
            totalPoints += s.points || 0;
            if ((s.progress || 0) > 20) active++;
            
            const status = (s.progress || 0) > 70 ? ['نشط', 'bg-emerald-500/20 text-emerald-300'] :
                          (s.progress || 0) > 30 ? ['متوسط', 'bg-amber-500/20 text-amber-300'] :
                          ['يحتاج متابعة', 'bg-rose-500/20 text-rose-300'];
            
            tbody.innerHTML += `
                <tr class="hover:bg-white/5 transition-all">
                    <td class="px-4 py-3">
                        <div class="flex items-center gap-2">
                            <img src="${s.photoURL || 'https://api.dicebear.com/7.x/avataaars/svg?seed=' + s.name}" class="w-8 h-8 rounded-full bg-white/10">
                            <div class="font-semibold text-xs">${s.name}</div>
                        </div>
                    </td>
                    <td class="px-4 py-3 text-xs text-slate-400">${s.email}</td>
                    <td class="px-4 py-3">
                        <div class="flex items-center gap-2">
                            <div class="w-16 bg-white/10 rounded-full h-1.5">
                                <div class="bg-blue-500 h-1.5 rounded-full" style="width: ${s.progress || 0}%"></div>
                            </div>
                            <span class="text-[10px]">${s.progress || 0}%</span>

function switchTeacherTab(tab) {
    document.querySelectorAll('.teacher-tab').forEach(t => {
        t.classList.remove('bg-white/10', 'text-white', 'font-semibold');
        t.classList.add('text-slate-400');
    });
    document.querySelector(`[data-tab="${tab}"]`).classList.add('bg-white/10', 'text-white', 'font-semibold');
    document.querySelector(`[data-tab="${tab}"]`).classList.remove('text-slate-400');
    
    document.querySelectorAll('.teacher-panel').forEach(p => p.classList.add('hidden'));
    document.getElementById('teacher-tab-' + tab).classList.remove('hidden');
}

function filterStudents() {
    const query = document.getElementById('student-search').value.toLowerCase();
    const rows = document.querySelectorAll('#teacher-students-table tr');
    rows.forEach(row => {
        const name = row.querySelector('.font-semibold')?.textContent.toLowerCase() || '';
        row.style.display = name.includes(query) ? '' : 'none';
    });
}
