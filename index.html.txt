"use client";

import React, { useEffect, useState } from "react";
import Link from "next/link";
import { useAuth } from "@/context/AuthContext";
import { GRADE_OPTIONS, getGradeLabel } from "@/lib/grades";
import MathText from "@/components/MathText";
import {
  BookOpen,
  FileText,
  MessageSquare,
  Sparkles,
  ArrowLeft,
  Users,
  Award,
  CheckCircle2,
  Download,
  Bell,
  ChevronLeft,
  GraduationCap
} from "lucide-react";

export default function HomePage() {
  const { user, setIsAuthModalOpen } = useAuth();
  const [announcements, setAnnouncements] = useState<any[]>([]);
  const [featuredPdfs, setFeaturedPdfs] = useState<any[]>([]);
  const [featuredForum, setFeaturedForum] = useState<any[]>([]);
  const [stats, setStats] = useState({
    totalStudents: 120,
    totalPdfs: 18,
    totalQuestions: 85,
    totalForum: 42,
  });

  useEffect(() => {
    // Fetch home data
    fetch("/api/announcements")
      .then((res) => res.json())
      .then((data) => {
        if (data.announcements) setAnnouncements(data.announcements.slice(0, 3));
      })
      .catch((err) => console.error(err));

    fetch("/api/pdfs")
      .then((res) => res.json())
      .then((data) => {
        if (data.pdfs) setFeaturedPdfs(data.pdfs.slice(0, 3));
      })
      .catch((err) => console.error(err));

    fetch("/api/forum?status=approved")
      .then((res) => res.json())
      .then((data) => {
        if (data.posts) setFeaturedForum(data.posts.slice(0, 3));
      })
      .catch((err) => console.error(err));

    fetch("/api/admin/stats")
      .then((res) => res.json())
      .then((data) => {
        if (data.stats) {
          setStats({
            totalStudents: Math.max(data.stats.totalStudents, 15),
            totalPdfs: Math.max(data.stats.totalPdfs, 6),
            totalQuestions: Math.max(data.stats.totalQuestions, 10),
            totalForum: Math.max(data.stats.totalForum, 8),
          });
        }
      })
      .catch((err) => console.error(err));
  }, []);

    <div className="space-y-16 pb-16 dir-rtl">
      
      {/* Hero Section */}

        {/* Background Decorative Math Shapes & Glows */}
        <div className="absolute top-10 right-10 w-96 h-96 bg-emerald-500/10 rounded-full blur-3xl pointer-events-none" />
        <div className="absolute bottom-10 left-10 w-96 h-96 bg-teal-500/10 rounded-full blur-3xl pointer-events-none" />
        
        {/* Floating LaTeX equations background */}
        <div className="absolute inset-0 opacity-5 font-mono text-emerald-400 text-sm select-none pointer-events-none overflow-hidden flex flex-wrap gap-8 p-6">
          <span>{"$\\lim_{x \\to 0} \\frac{\\sin x}{x} = 1$"}</span>
          <span>{"$E = mc^2$"}</span>
          <span>{"$a^2 + b^2 = c^2$"}</span>
          <span>{"$\\int x^n dx = \\frac{x^{n+1}}{n+1}$"}</span>
          <span>{"$\\cos^2 \\theta + \\sin^2 \\theta = 1$"}</span>
        </div>

        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 relative z-10">
          <div className="grid grid-cols-1 lg:grid-cols-12 gap-12 items-center">
            
            {/* Hero Text */}
            <div className="lg:col-span-7 space-y-6 text-right">
              
              <div className="inline-flex items-center gap-2 px-3.5 py-1.5 rounded-full bg-emerald-500/10 border border-emerald-500/30 text-emerald-400 text-xs font-bold shadow-sm">
                <Sparkles className="w-4 h-4" />
                <span>المنصة الأولى المتخصصة لطلاب المرحلة الإعدادية والثانوية</span>
              </div>

              <h1 className="text-4xl sm:text-5xl lg:text-6xl font-black text-white leading-tight">
                طريقك للتميز والدرجة النهائية مع <br />
                <span className="bg-gradient-to-r from-emerald-400 via-teal-300 to-cyan-400 bg-clip-text text-transparent">
                  مستر وليد عربان
                </span>
              </h1>

              <p className="text-base sm:text-lg text-slate-300 leading-relaxed max-w-2xl">
                منصة تعليمية متكاملة توفر لك أحدث مذكرات الشرح والملخصات صيغة PDF، وبنك أسئلة تفاعلي بإجابات نموذجية خطوة بخطوة، بالإضافة لـ <strong className="text-emerald-400">منتدى أسئلة الطلاب</strong> للإجابة المباشرة المعتمدة من مستر وليد عربان.
              </p>

              {/* Action Buttons */}
              <div className="flex flex-wrap items-center gap-4 pt-2">
                {user ? (
                  <Link
                    href="/pdfs"
                    className="py-3.5 px-7 bg-gradient-to-r from-emerald-500 to-teal-500 hover:from-emerald-400 hover:to-teal-400 text-slate-950 font-black rounded-2xl shadow-xl shadow-emerald-500/25 flex items-center gap-2 transition-all hover:scale-[1.02]"
                  >
                    <BookOpen className="w-5 h-5" />
                    <span>تصفح المذكرات الخاصة بصفك</span>
                  </Link>
                ) : (
                  <button
                    onClick={() => setIsAuthModalOpen(true)}
                    className="py-3.5 px-7 bg-gradient-to-r from-emerald-500 to-teal-500 hover:from-emerald-400 hover:to-teal-400 text-slate-950 font-black rounded-2xl shadow-xl shadow-emerald-500/25 flex items-center gap-2 transition-all hover:scale-[1.02]"
                  >
                    <Users className="w-5 h-5" />
                    <span>انضم الآن بحساب جوجل أو طالب جديد</span>
                  </button>
                )}

                <Link
                  href="/forum"
                  className="py-3.5 px-6 bg-slate-900 hover:bg-slate-800 text-slate-100 font-bold rounded-2xl border border-slate-700 flex items-center gap-2 transition-all"
                >
                  <MessageSquare className="w-5 h-5 text-emerald-400" />
                  <span>اسأل مستر وليد في المنتدى</span>
                </Link>
              </div>

              {/* Quick Key Highlights */}
              <div className="grid grid-cols-3 gap-4 pt-6 border-t border-slate-800/80 max-w-xl">
                <div className="text-right">
                  <div className="text-2xl font-black text-white">+{stats.totalStudents}</div>
                  <div className="text-xs text-slate-400">طالب مسجل</div>
                </div>
                <div className="text-right">
                  <div className="text-2xl font-black text-emerald-400">+{stats.totalPdfs}</div>
                  <div className="text-xs text-slate-400">مذكرة PDF مجانية</div>
                </div>
                <div className="text-right">
                  <div className="text-2xl font-black text-teal-400">+{stats.totalQuestions}</div>
                  <div className="text-xs text-slate-400">سؤال تفاعلي</div>
                </div>
              </div>

            </div>

            {/* Hero Card / Teacher Badge */}
            <div className="lg:col-span-5 relative">
              <div className="relative bg-gradient-to-b from-slate-900 to-slate-950 border border-slate-800 p-6 sm:p-8 rounded-3xl shadow-2xl space-y-6">
                
                <div className="flex items-center gap-4 border-b border-slate-800 pb-5">
                  <div className="w-16 h-16 rounded-2xl bg-gradient-to-tr from-emerald-500 via-teal-400 to-cyan-500 p-0.5 shadow-lg">
                    <div className="w-full h-full bg-slate-950 rounded-[14px] flex items-center justify-center font-black text-3xl text-emerald-400">
                      π
                    </div>
                  </div>
                  <div>
                    <h3 className="text-xl font-bold text-white">مستر وليد عربان</h3>
                    <p className="text-xs text-emerald-400 font-semibold">مدرس أول الرياضيات للمرحلتين الإعدادية والثانوية</p>
                    <p className="text-xs text-slate-400">mr. waleed orban - math platform</p>
                  </div>
                </div>

                {/* Math Preview Feature Box */}
                <div className="bg-slate-950/80 border border-emerald-500/20 rounded-2xl p-4 text-xs space-y-2">
                  <div className="flex items-center justify-between text-emerald-400 font-bold">
                    <span>💡 معادلة اليوم الشائعة:</span>
                    <span className="bg-emerald-500/10 px-2 py-0.5 rounded">2 ثانوي</span>
                  </div>
                  <div className="py-2 text-slate-200">
                    <MathText text="قانون جيب التمام: $c^2 = a^2 + b^2 - 2ab \cos(C)$" />
                  </div>
                  <p className="text-[11px] text-slate-400">شرح وتطبيق إثبات النظريات متوفر في بنك الأسئلة والمذكرات.</p>
                </div>

                <div className="space-y-2 text-xs text-slate-300">
                  <div className="flex items-center gap-2">
                    <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />
                    <span>مذكرات شاملة ومجانية بصيغة PDF للطباعة والقراءة.</span>
                  </div>
                  <div className="flex items-center gap-2">
                    <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />
                    <span>منتدى أسئلة يتضمن ردوداً موثقة من مستر وليد بنفسه.</span>
                  </div>
                  <div className="flex items-center gap-2">
                    <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />
                    <span>متابعة خاصة للنتائج واختبارات قياس المستوى.</span>
                  </div>
                </div>

              </div>
            </div>

          </div>
        </div>
      </section>

      {/* Announcements Section */}
      {announcements.length > 0 && (
        <section className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
          <div className="bg-gradient-to-r from-emerald-950/40 via-slate-900 to-teal-950/40 border border-emerald-500/30 rounded-2xl p-4 sm:p-6 shadow-lg">
            <div className="flex items-center gap-3 mb-4">
              <div className="p-2 bg-emerald-500/20 text-emerald-400 rounded-xl">
                <Bell className="w-5 h-5" />
              </div>
              <div>
                <h3 className="font-extrabold text-lg text-white">آخر التنبيهات والإعلانات العامة</h3>
                <p className="text-xs text-slate-400">تابع أحدث المراجعات ومواعيد رفع المذكرات من مستر وليد</p>
              </div>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
              {announcements.map((item) => (
                <div
                  key={item.id}
                  className="bg-slate-900/90 border border-slate-800 p-4 rounded-xl space-y-2 hover:border-emerald-500/40 transition-colors"
                >
                  <div className="flex items-center justify-between">
                    <span className="text-xs font-bold text-emerald-400 bg-emerald-500/10 px-2 py-0.5 rounded">
                      {getGradeLabel(item.grade)}
                    </span>
                    <span className="text-[10px] text-slate-500">
                      {new Date(item.createdAt).toLocaleDateString("ar-EG")}
                    </span>
                  </div>
                  <h4 className="font-bold text-sm text-slate-100">{item.title}</h4>
                  <p className="text-xs text-slate-400 line-clamp-2">{item.content}</p>
                </div>
              ))}
            </div>
          </div>
        </section>
      )}

      {/* Grade Levels Selection Section */}
      <section className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 space-y-8">
        <div className="text-center max-w-2xl mx-auto space-y-2">
          <h2 className="text-2xl sm:text-3xl font-black text-white">
            اختر صفك الدراسي وابدأ التعلم الآن
          </h2>
          <p className="text-sm text-slate-400">
            تغطية شاملة لمناهج الرياضيات للصفوف من الأول الإعدادي حتى الثاني الثانوي
          </p>
        </div>

        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
          {GRADE_OPTIONS.map((grade) => (
            <div
              key={grade.id}
              className="bg-slate-900/80 border border-slate-800 hover:border-emerald-500/50 rounded-2xl p-6 transition-all duration-200 hover:shadow-xl hover:shadow-emerald-500/10 group flex flex-col justify-between"
            >
              <div className="space-y-4">
                <div className="flex items-center justify-between">
                  <div className="w-12 h-12 rounded-xl bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 flex items-center justify-center group-hover:bg-emerald-500 group-hover:text-slate-950 transition-colors">
                    <GraduationCap className="w-6 h-6" />
                  </div>
                  <span className="text-xs font-bold bg-slate-800 text-slate-300 px-3 py-1 rounded-full">
                    {grade.stage === "prep" ? "المرحلة الإعدادية" : "المرحلة الثانوية"}
                  </span>
                </div>

                <div>
                  <h3 className="text-lg font-extrabold text-white group-hover:text-emerald-400 transition-colors">
                    {grade.label}
                  </h3>
                  <div className="flex flex-wrap gap-1.5 mt-2">
                    {grade.branches.map((b, i) => (
                      <span key={i} className="text-[11px] bg-slate-950 text-slate-400 border border-slate-800 px-2 py-0.5 rounded-md">
                        {b}
                      </span>
                    ))}
                  </div>
                </div>
              </div>

              <div className="pt-6 mt-4 border-t border-slate-800/80 flex items-center justify-between text-xs">
                <Link
                  href={`/pdfs?grade=${grade.id}`}
                  className="font-bold text-emerald-400 hover:text-emerald-300 flex items-center gap-1 group-hover:translate-x-[-4px] transition-transform"
                >
                  <span>عرض المذكرات</span>
                  <ChevronLeft className="w-4 h-4" />
                </Link>
                <Link
                  href={`/questions?grade=${grade.id}`}
                  className="font-bold text-teal-400 hover:text-teal-300 flex items-center gap-1"
                >
                  <span>الاختبارات</span>
                </Link>
              </div>
            </div>
          ))}
        </div>
      </section>

      {/* Featured PDFs Section */}
      <section className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 space-y-6">
        <div className="flex items-center justify-between">
          <div>
            <h2 className="text-2xl font-black text-white flex items-center gap-2">
              <FileText className="w-6 h-6 text-emerald-400" />
              <span>أحدث مذكرات وملفات الـ PDF</span>
            </h2>
            <p className="text-xs text-slate-400 mt-1">
              مذكرات من إعداد مستر وليد عربان جاهزة للتحميل المباشر
            </p>
          </div>

          <Link
            href="/pdfs"
            className="text-xs font-bold text-emerald-400 hover:text-emerald-300 bg-emerald-500/10 border border-emerald-500/20 px-4 py-2 rounded-xl flex items-center gap-1 transition-all"
          >
            <span>جميع المذكرات</span>
            <ArrowLeft className="w-4 h-4" />
          </Link>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
          {featuredPdfs.map((pdf) => (
            <div
              key={pdf.id}
              className="bg-slate-900 border border-slate-800 rounded-2xl p-5 hover:border-emerald-500/40 transition-all flex flex-col justify-between"
            >
              <div className="space-y-3">
                <div className="flex items-center justify-between text-xs">
                  <span className="bg-emerald-500/10 text-emerald-400 font-bold px-2.5 py-0.5 rounded-lg border border-emerald-500/20">
                    {getGradeLabel(pdf.grade)}
                  </span>
                  <span className="text-slate-400 bg-slate-800 px-2 py-0.5 rounded text-[11px]">
                    {pdf.branch}
                  </span>
                </div>

                <h3 className="font-bold text-base text-white leading-snug line-clamp-2">
                  {pdf.title}
                </h3>

                <p className="text-xs text-slate-400 line-clamp-2">{pdf.description}</p>
              </div>

              <div className="pt-4 mt-4 border-t border-slate-800 flex items-center justify-between text-xs text-slate-400">
                <div className="flex items-center gap-3">
                  <span>📄 {pdf.pageCount} صفحة</span>
                  <span>💾 {pdf.fileSize}</span>
                </div>
                <Link
                  href="/pdfs"
                  className="p-2 bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-bold rounded-lg flex items-center gap-1 transition-colors"
                >
                  <Download className="w-3.5 h-3.5" />
                  <span>تحميل</span>
                </Link>
              </div>
            </div>
          ))}
        </div>
      </section>

      {/* Featured Forum Q&A Section */}
      <section className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 space-y-6">
        <div className="flex items-center justify-between">
          <div>
            <h2 className="text-2xl font-black text-white flex items-center gap-2">
              <MessageSquare className="w-6 h-6 text-emerald-400" />
              <span>منتدى أسئلة الطلاب مع مستر وليد</span>
            </h2>
            <p className="text-xs text-slate-400 mt-1">
              أسئلة الطلاب الموجّهة للمعلّم مع الإجابة النموذجية المعتمدة
            </p>
          </div>

          <Link
            href="/forum"
            className="text-xs font-bold text-emerald-400 hover:text-emerald-300 bg-emerald-500/10 border border-emerald-500/20 px-4 py-2 rounded-xl flex items-center gap-1 transition-all"

            <span>دخول المنتدى وطرح سؤال</span>

            <div

                    className="w-8 h-8 rounded-full bg-slate-800"
                  />

                  <span>تمت الإجابة والاعتماد</span>
                </span>
              </div>

              <div>
                <h3 className="font-extrabold text-base text-white mb-1">{post.title}</h3>
                <div className="text-xs sm:text-sm text-slate-300 leading-relaxed">
                  <MathText text={post.content} />
                </div>
              </div>

              {post.teacherAnswer && (
                <div className="bg-gradient-to-r from-emerald-950/50 to-slate-950 border border-emerald-500/40 rounded-xl p-4 text-xs space-y-2">
